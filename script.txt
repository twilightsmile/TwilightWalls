// Category Change
function showCategory(category){

    currentCategory = category;

    setActiveButton(category + "Btn");

    // Title change
    document.getElementById("title").innerHTML =
    "🌙 " + category.charAt(0).toUpperCase() + category.slice(1) + " Wallpapers";

    // Hide all cards
    document.querySelectorAll(".card").forEach(card=>{
        card.style.display="none";
    });

    // Show selected category
    document.querySelectorAll("." + category).forEach(card=>{
        card.style.display="block";
    });

}

function setActiveButton(buttonId){

    const buttons = [
        "homeBtn",
        "carsBtn",
        "natureBtn",
        "animeBtn",
        "spaceBtn",
        "amoledBtn",
        "favBtn",
        "downloadBtn",
        "settingsBtn"
    ];

    buttons.forEach(id=>{
        const btn=document.getElementById(id);

        if(btn){
            btn.classList.remove("active");
        }
    });

    const active=document.getElementById(buttonId);

    if(active){
        active.classList.add("active");
    }

}

function toggleFavorite(btn){

    let card = btn.closest(".card");
    let title = card.querySelector("h3").innerText;
    let image = card.querySelector("img").getAttribute("src");

    let index = favorites.findIndex(item => item.title === title);

    if(index !== -1){

        favorites.splice(index,1);

        btn.classList.remove("active");
        btn.innerHTML="🤍 Favorite";

    }else{

        favorites.push({
            title:title,
            image:image
        });

        btn.classList.add("active");
        btn.innerHTML="❤️ Favorite";

    }

    localStorage.setItem("favorites", JSON.stringify(favorites));
    
renderFavorites();
}

function searchWallpapers(){

    let input = document.getElementById("searchBox").value.toLowerCase();

    let cards = document.querySelectorAll(".card");

    cards.forEach(card=>{

        let title = card.querySelector("h3");

        if(title){

            if(title.innerHTML.toLowerCase().includes(input)){
                card.style.display="block";
            }else{
                card.style.display="none";
            }

        }

    });

}
function renderFavorites(){

    let gallery = document.getElementById("favoritesGallery");

    let message = document.getElementById("noFavMessage");

    gallery.innerHTML="";

    if(favorites.length===0){

        message.style.display="block";

        return;

    }

    message.style.display="none";

    favorites.forEach(item=>{

        gallery.innerHTML += `

        <div class="card">

            <img src="${item.image}"
                 onclick="openImage('${item.image}')">

            <h3>${item.title}</h3>

            <button class="fav-btn active"
            onclick="removeFavorite('${item.title}')">

            ❤️ Remove

            </button>

            <a href="${item.image}" download>

                <button>⬇ Download</button>

            </a>

        </div>

        `;

    });

}
function removeFavorite(title){

    favorites = favorites.filter(item=>item.title!==title);

    localStorage.setItem("favorites",JSON.stringify(favorites));

    renderFavorites();

}

window.onscroll = function(){

    let btn = document.getElementById("topBtn");

    if(document.documentElement.scrollTop > 300){

        btn.style.display = "block";

    }else{

        btn.style.display = "none";

    }

}
